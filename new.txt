server {
    listen 443 ssl;
    server_name uat-ers.religare.in 10.91.41.16;

    ssl_certificate /etc/ssl/religare/religare.in.crt;
    ssl_certificate_key /etc/ssl/religare/religare.in.key;

    root /opt/ER_Linux/frontend/build;
    index index.html;

    # Proxy API requests to backend
    location /api/ {
        proxy_pass http://localhost:5000;  # Change to your backend port
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    # Serve static files directly
    location /static/ {
        alias /opt/ER_Linux/frontend/build/static/;
        expires 30d;
        add_header Cache-Control "public, max-age=2592000";
    }

    # Serve other assets (favicon, manifest, etc.)
    location ~* \.(?:ico|css|js|gif|jpe?g|png|woff2?|ttf|svg|eot|otf|json)$ {
        try_files $uri =404;
        expires 30d;
        add_header Cache-Control "public, max-age=2592000";
    }

    # All other requests serve React's index.html (for client-side routing)
    location / {
        try_files $uri /index.html;
    }
}

server {
    listen 80;
    server_name uat-ers.religare.in 10.91.41.16;
    return 301 https://$host$request_uri;
}
